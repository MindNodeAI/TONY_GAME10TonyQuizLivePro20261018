Gemini AI 設定與 Firestore 安全規則

系統優先呼叫 gemini-4-hydrogen，並自動備援至 gemini-3.8-flash 與 gemini-3.5-flash。

✅ 已內建連線 Firebase 專案：game10-tonyquiz

會員認證 (Authentication) 與即時互動房間 (Cloud Firestore) 皆已自動綁定。

請複製以下規則貼至 Firebase Console > Firestore Database > 規則 (Rules) 並發布：

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // 判斷是否已完成登入
    function isSignedIn() {
      return request.auth != null;
    }

    // 判斷是否為系統創作者最高權限 (TONY 楊文澤老師)
    function isCreator() {
      return isSignedIn() &&
        request.auth.token.email != null &&
        request.auth.token.email.lower() == 'mindnodeai@gmail.com';
    }

    // 判斷是否為完成信箱驗證之有效會員 (或創作者)
    function isVerifiedMember() {
      return isCreator() || (isSignedIn() && request.auth.token.email_verified == true);
    }

    // 判斷是否具備 Admin 管理員角色
    function isAdmin() {
      return isCreator() || (
        isVerifiedMember() &&
        exists(/databases/$(database)/documents/tonyquiz_users/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/tonyquiz_users/$(request.auth.uid)).data.role == 'admin'
      );
    }

    // 1. 會員資料與 RBAC 權限表
    match /tonyquiz_users/{userId} {
      // 本人或管理員可讀取會員資料
      allow read: if isSignedIn() && (request.auth.uid == userId || isAdmin());

      // 註冊建立個人檔案：本人可建立，且禁止一般人偽造自設為 admin
      allow create: if isSignedIn() && request.auth.uid == userId && (
        isCreator() || request.resource.data.get('role', 'user') == 'user'
      );

      // 更新資料：
      // (a) 本人登入更新次數與歷史紀錄時，不可竄改自身 role 權限
      // (b) 管理員可升降一般會員權限，但永久禁止降級創作者帳號
      allow update: if isSignedIn() && (
        (request.auth.uid == userId &&
          request.resource.data.role == resource.data.role) ||
        (isAdmin() &&
          resource.data.get('email', '').lower() != 'mindnodeai@gmail.com') ||
        isCreator()
      );

      // 刪除帳號：僅管理員可執行，且永久禁止刪除創作者帳號
      allow delete: if isAdmin() &&
        resource.data.get('email', '').lower() != 'mindnodeai@gmail.com';
    }

    // 2. 課堂即時互動房間 (支援百人同時搶答、文字雲與匿名提問牆)
    match /tonyquiz_rooms/{roomPin} {
      allow read, create, update: if isVerifiedMember();
      allow delete: if isAdmin();
    }
  }
}
